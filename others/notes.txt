// HWDB1.1 (Hugging Face)

Page:
https://huggingface.co/datasets/OrkaZeta/HWDB1/tree/main

download in shell:

New-Item -ItemType Directory -Force D:\Main\DS_data\HWDB1.1 | Out-Null
cd D:\Main\DS_data\HWDB1.1
echo download
curl.exe -L --retry 5 -C - -O https://huggingface.co/datasets/OrkaZeta/HWDB1/resolve/main/Gnt1.1Test.zip
curl.exe -L --retry 5 -C - -O https://huggingface.co/datasets/OrkaZeta/HWDB1/resolve/main/Gnt1.1TrainPart1.zip
curl.exe -L --retry 5 -C - -O https://huggingface.co/datasets/OrkaZeta/HWDB1/resolve/main/Gnt1.1TrainPart2.zip
echo checksum
Get-FileHash Gnt1.1Test.zip,Gnt1.1TrainPart1.zip,Gnt1.1TrainPart2.zip -Algorithm SHA256 | Format-Table Hash,Path -AutoSize
Expand-Archive .\Gnt1.1Test.zip -DestinationPath .\HWDB1.1tst_gnt -Force
Expand-Archive .\Gnt1.1TrainPart1.zip -DestinationPath .\HWDB1.1trn_gnt -Force
Expand-Archive .\Gnt1.1TrainPart2.zip -DestinationPath .\HWDB1.1trn_gnt -Force
